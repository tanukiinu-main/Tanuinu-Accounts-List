async function renderList() { const userListEl = document.getElementById('user-list'); const nameInput = document.getElementById('name-input'); const hashName = decodeURIComponent(window.location.hash.replace('#', '').trim()); if (document.activeElement !== nameInput) { nameInput.value = hashName; } const myInputHash = hashName ? await getSha256(hashName) : ''; const fragment = document.createDocumentFragment(); HASH_LIST.forEach((hash) => { const cleanHash = hash.toLowerCase().trim(); const isMatched = (myInputHash !== '' && myInputHash === cleanHash); const displayName = isMatched ? hashName : cleanHash; const accountData = workerStatusData.get(cleanHash) || { status: 'offline', location: '' }; const isOnline = accountData.status !== 'offline'; const loc = accountData.location; // HTML側でパブリック判定を行う const isPublicWorld = loc.startsWith('wrld_') && !loc.includes('~private') && !loc.includes('~friends') && !loc.includes('~hidden') && !loc.includes('~group'); // 表示条件判定： // ・管理者モードなら、生データにlocationがあれば無条件表示 // ・一般モードなら、(自分のハッシュ一致) かつ (パブリックワールド) の場合のみ表示 let showWorldInfo = false; if (loc !== '') { if (isAdminMode) { showWorldInfo = true; // 管理者はプライベート・フレンド限定インスタンス含めて全表示 } else if (isMatched && isPublicWorld) { showWorldInfo = true; // 一般ユーザーは自分のアカウントかつパブリックのみ } } const li = document.createElement('li'); li.innerHTML = ` ${displayName}
${isOnline ? accountData.status : 'オフライン'} ${showWorldInfo ? `${loc}` : ''}
`; fragment.appendChild(li); }); userListEl.innerHTML = ''; userListEl.appendChild(fragment); }
